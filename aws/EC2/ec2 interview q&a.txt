import boto3

iam = boto3.client('iam')

def audit_ec2_roles():
    roles = iam.list_roles()

        's3:*', 'dynamodb:*', 'ec2:*', 'iam:*', 'sts:AssumeRole'
    ]

                RoleName=role['RoleName']
            )

                    VersionId=iam.get_policy(
                        PolicyArn=policy['PolicyArn']
                    )['Policy']['DefaultVersionId']

                for statement in policy_version['PolicyVersion']['Document']['Statement']:
                    if statement.get('Effect') == 'Allow':
                        actions = statement.get('Action', [])
                        if isinstance(actions, str):
                            actions = [actions]

                            if any(perm in action for perm in dangerous_permissions):
                                print(f"ALERT: Role {role['RoleName']} has overly permissive action: {action}")

    --listener-arns arn:aws:elasticloadbalancing:... \
    --query 'Listeners[0].DefaultActions[0].TargetGroupArn' \

else
    CURRENT_COLOR="green"

    NEW_TG_ARN="arn:aws:elasticloadbalancing:.../app-blue"

    --min-size 2 \
    --max-size 10 \
    --desired-capacity 5

    --auto-scaling-group-name app-$NEW_COLOR

sleep 60

HEALTHY_COUNT=$(aws elbv2 describe-target-health \
    --target-group-arn $NEW_TG_ARN \
    --query 'TargetHealthDescriptions[?TargetHealth.State==`healthy`] | length(@)')

    exit 1
fi

    --listener-arn arn:aws:elasticloadbalancing:... \
    --default-actions Type=forward,TargetGroupArn=$NEW_TG_ARN

sleep 600

# If successful, scale down old environment
read -p "Deployment successful? Scale down $CURRENT_COLOR? (y/n) " -n 1 -r
